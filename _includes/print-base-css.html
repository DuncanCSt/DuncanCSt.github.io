/* Shared by every printed document (resume and cover letters) so they read as one set. Document-specific rules stay in the page. */ /* Self-hosted static TTFs, not the Google Fonts CDN. Chrome embeds the CDN copy as Type3 (glyph drawings) rather than a real text font, which some resume parsers cannot read. Static local files embed as Type0. Also makes the PDF build work offline. */ @font-face { font-family: "Muli"; font-weight: 400; font-style: normal; src: url("/assets/fonts/muli-400.ttf") format("truetype"); } @font-face { font-family: "Muli"; font-weight: 800; font-style: normal; src: url("/assets/fonts/muli-800.ttf") format("truetype"); } :root { --accent: #2d6a4f; --muted: #868e96; --edge-buffer: 0.1in; } * { box-sizing: border-box; } html, body { margin: 0; padding: 0; background: #fff; color: #212529; } body { font-family: "Muli", -apple-system, sans-serif; font-size: 11pt; line-height: 1.4; /* Keep this: Muli's fi/fl ligature glyphs carry no reverse mapping, so with ligatures on, "Snowflake" extracts as "Snow ake" and "Certificate" as "Certi cate" — the letters are silently lost, and any keyword containing them with it. */ font-variant-ligatures: none; } .page { max-width: 8.27in; margin: 0 auto; padding: 0.5in 0.6in; } /* scripts/build-pdf.js supplies the page margins per sheet, so the padding would double up here. */ @media print { .page { padding: 0; max-width: none; } } h1, h2, h3 { font-family: "Barlow Semi Condensed", sans-serif; text-transform: uppercase; margin: 0; } h1 { font-size: 32pt; font-weight: 700; letter-spacing: 0.02em; } h1 .last { color: var(--accent); } h3 { font-size: 13pt; } .contact { color: #495057; font-size: 10pt; margin: 0.2rem 0 0.6rem; } /* Second contact line (website, LinkedIn, GitHub). Full URLs are long, so this runs a touch smaller and tightens the top margin to keep the header block the same overall height. */ /* All three URLs plus their icons fit one line at 9pt. If a link ever grows, this is the first thing to drop back to 8.5pt — wrapping the last URL onto its own line looks far worse. */ .contact-links { font-size: 9pt; margin: -0.45rem 0 0.6rem; } /* Inline SVG marks, sized to the cap height of the line. They carry no text layer of their own, so the URLs beside them still extract exactly as they did before the icons existed. */ .contact-links .ico { width: 8.5px; height: 8.5px; vertical-align: -0.08em; margin-right: 0.22em; } /* The site URL is a real anchor so it is clickable in the PDF, but it reads as part of the contact line, not as accent-coloured. */ .contact a { color: inherit; } a { color: var(--accent); text-decoration: none; }
